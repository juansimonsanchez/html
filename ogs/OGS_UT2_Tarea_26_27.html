25. Número de Palabras ¶

Algoritmo NumeroPalabras
// Determina el número de palabras en un texto contando los espacios en blanco que dividen los términos gramaticales.
Definir frase, letra Como Cadena;
Definir i, palabras, longitudFrase Como Entero;

Escribir "Escriba una frase:";
Leer frase;

longitudFrase <- Longitud(frase);
palabras <- 0;

Si longitudFrase > 0 Entonces
// Partimos de que hay al menos 1 palabra si la frase no está vacía
palabras <- 1;
Para i <- 1 Hasta longitudFrase Con Paso 1 Hacer
letra <- Subcadena(frase, i, i);
// Si el carácter actual es un espacio y el siguiente no lo es (evitando dobles espacios)
Si letra == " " Y (i < longitudFrase Y Subcadena(frase, i + 1, i + 1) <> " ") Entonces
palabras <- palabras + 1;
FinSi
FinPara
FinSi

Escribir "Número de palabras estimadas: ", palabras;
FinAlgoritmo

27. Cruce Seguro (Algoritmo estructurado del Diagrama de Flujo) ¶

Algoritmo CruceSeguroAvenida
// Descripción del comportamiento lógico y secuencial recomendado para modelar mediante diagrama de flujo cómo cruzar una avenida con semáforo peatonal de forma correcta.
Definir colorSemaforo, vehiculosAproximandose Como Cadena;
Definir cruceExitoso Como Logico;

cruceExitoso <- Falso;

Repetir
Escribir "Estás en la acera. Observas el semáforo de peatones.";
Escribir "¿De qué color es la luz del semáforo peatonal? (VERDE / ROJO):";
Leer colorSemaforo;
colorSemaforo <- Mayusculas(colorSemaforo);

Si colorSemaforo == "ROJO" Entonces
Escribir "ALERTA DETENCIÓN: No cruzar. Espera seguro en la acera.";
Escribir "--------------------------------------------------------";
Sino
Si colorSemaforo == "VERDE" Entonces
Escribir "¿Se aproximan vehículos ignorando la señal o de emergencia? (SI / NO):";
Leer vehiculosAproximandose;
vehiculosAproximandose <- Mayusculas(vehiculosAproximandose);

Si vehiculosAproximandose == "NO" Entonces
Escribir "Cruzas la calzada de forma ágil y con precaución.";
cruceExitoso <- Verdadero;
Sino
Escribir "ALERTA DETENCIÓN: Vehículo peligroso aproximándose. Espera.";
Escribir "--------------------------------------------------------";
FinSi
Sino
Escribir "Estado desconocido. Mantente a salvo en la acera.";
FinSi
FinSi
Hasta Que cruceExitoso == Verdadero;

Escribir "¡Has cruzado la avenida con total seguridad!";
FinAlgoritmo
